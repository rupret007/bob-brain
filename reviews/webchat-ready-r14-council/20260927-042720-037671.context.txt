# Web Bob Round 14 — design note (research-first)

**Borrow**
- R8/R13 detailed live dashboard contract (`running_jobs[]` log tail, scores, roles, MiniMax cap) — extend Jobs tab dl grid with per-job snippets + compact scores (Jeff 02:03).
- One-tap quick chips that submit (not only prefill) for status / round / council — pairs with existing approve cards (Jeff 02:19, 04:18).
- Grok Bot call barge-in: cancel TTS on interim speech (MDN `SpeechRecognition` interimResults).
- Self-improve loop visibility: streak of OK ticks + `loop_ready` flag for Settings/Jobs (Jeff 00:42).

**Reject**
- MiniMax music API / signups (Jeff 02:00).
- Grok as worker.
- Auto-confirm jobs without approve card (safety).

**Why**
- Lowest R13 scores: self-improve loop (9.28), run jobs from chat (9.6), conductor parity (9.58), voice (9.52), music (9.65).
- Jobs tab must mirror Models dashboard detail, not only summary rows.

**Sources:** `WEBCHAT-READY-R13-RESULT.md`, `WEBCHAT-READY-R8-RESULT.md`, Apple HIG tab bar, MDN Web Speech API, `ALIGNMENT-DESIGN.md`.

  function renderMarkdown(text) {
    let html = escapeHtml(text);
    html = html.replace(/```([\s\S]*?)```/g, (_, code) => {
      const inner = code.trim();
      return `<pre class="code-block" tabindex="0"><button type="button" class="copy-code" aria-label="Copy code">Copy</button><code>${inner}</code></pre>`;
    });
    html = html.replace(/`([^`]+)`/g, "<code>$1</code>");
    html = html.replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g, '<a href="$2" rel="noopener noreferrer" target="_blank">$1</a>');
    html = html.replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>");
    html = html.replace(/^### (.+)$/gm, "<h4 class=\"md-h\">$1</h4>");
    html = html.replace(/^## (.+)$/gm, "<h3 class=\"md-h\">$1</h3>");
    html = html.replace(/(?:^|\n)(\d+\. .+(?:\n\d+\. .+)*)/g, (block) => {
      const items = block.trim().split("\n").map((line) => `<li>${line.replace(/^\d+\.\s+/, "")}</li>`).join("");
      return `<ol>${items}</ol>`;
    });
    html = html.replace(/(?:^|\n)([-*] .+(?:\n[-*] .+)*)/g, (block) => {
      const items = block.trim().split("\n").map((line) => `<li>${line.replace(/^[-*]\s+/, "")}</li>`).join("");
      return `<ul>${items}</ul>`;
    });
    html = html.replace(/\n/g, "<br>");
    return html;
  }

  let callSession = null;
  async function drainCallQueue(session) {
    if (!session || session.drainRunning || callSession !== session) return;
    session.drainRunning = true;
    try {
      while (callSession === session && session.queue.length) {
        const { transcript, heardAt } = session.queue.shift();
        if (heardAt !== session.lastHeardAt) continue;
        await handleCallUtterance(transcript, session.id);
      }
    } finally {
      if (session === callSession) session.drainRunning = false;
    }
  }
  function enqueueCallUtterance(transcript, session) {
    if (!session || callSession !== session) return;
    const norm = transcript.trim().toLowerCase();
    if (norm.length < 3) return;
    const last = session.queue.at(-1);
    if (last && last.transcript.trim().toLowerCase() === norm) return;
    const heardAt = Date.now();
    session.lastHeardAt = heardAt;
    session.queue.push({ transcript: transcript.trim(), heardAt });
    drainCallQueue(session);
  }

    });
  }

  async function handleCallUtterance(transcript, sessionId) {
    if (!callSession || callSession.id !== sessionId || !transcript.trim()) return;
    if (callSession.busy) return;
    callSession.busy = true;
    appendCallLine("you", transcript);
    $("call-status").textContent = "Bob is thinking…";
    speechSynthesis.cancel();
    try {
      const requestId = crypto.randomUUID();
      const payload = {
        conversation_id: current,
        message: transcript,
        provider: $("provider-choice").value,
        request_id: requestId,
      };
      await api("/api/messages", payload);
      const value = await waitForReply(current, requestId);
      if (!callSession || callSession.id !== sessionId) return;
      const msg = value.messages.find((m) => m.role === "assistant" && m.request_id === requestId);
      const body = msg?.content || "No reply yet.";
      appendCallLine("bob", body);
      $("call-status").textContent = "Listening… speak anytime (tap End to stop).";
      await speakCallReply(body);
    } catch (error) {
      $("call-status").textContent = error.message || "Call error";
    } finally {
      if (callSession?.id === sessionId) callSession.busy = false;
    }
  }

        const row = event.results[index];

      else if (event.error !== "aborted") $("call-status").textContent = "Mic error — try again or type in chat.";
    };
    rec.onend = () => {
      if (callSession?.recognition === rec) {
        try { rec.start(); } catch (_) { /* restart loop while call open */ }
      }
    };
    try { rec.start(); } catch (_) {
      stopLiveCall();
      $("notice").textContent = "Could not start live call.";
    }
  });

def private_write(path, value):
    temporary = path.with_name("." + path.name + "." + secrets.token_hex(6))
    try:
        with open(os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), "w") as handle:
            handle.write(value)
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(temporary, path)
        descriptor = os.open(path.parent, os.O_RDONLY)
        try:
            os.fsync(descriptor)
        finally:
            os.close(descriptor)
    finally:
        temporary.unlink(missing_ok=True)


def password_at(folder):
    path = folder / ".password"
    try:
        descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    except FileExistsError:
        descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
        with os.fdopen(descriptor) as handle:
            if not stat.S_ISREG(os.fstat(handle.fileno()).st_mode):
                raise RuntimeError("Password path must be a regular file")
            os.fchmod(handle.fileno(), 0o600)
            password = handle.read().strip()
        if len(password) < 24:
            raise RuntimeError("Password file must contain at least 24 characters")
        return password
    with os.fdopen(descriptor, "w") as handle:
        password = secrets.token_urlsafe(24)
        handle.write(password + "\n")
        handle.flush()
        os.fsync(handle.fileno())
    return password

        self.state.mkdir(mode=0o700, exist_ok=True)

        try:
            state_ledger.purge_drill_open_asks(self.root)
        except OSError:
            pass
        self.lock = threading.RLock()
        self.sessions, self.conversations = {}, {}
        self.overlay_roots: dict[str, Path] = {}
        self.overlay_pins: dict[str, Path] = {}
        self.failures = deque()
        self.workers = {}
        self.closing, self.queue_thread = threading.Event(), None
        self.providers = providers or Providers(
            root, self.redact, overlay_roots=self.overlay_roots, overlay_pins=self.overlay_pins,
        )
        for path in self.state.glob("*.json"):
            if not UUID.fullmatch(path.stem):
                continue
            # Corrupt state fails startup visibly; never silently discard history.
            value = json.loads(path.read_text())
            if value.get("id") != path.stem or not isinstance(value.get("messages"), list):
                raise RuntimeError("Invalid conversation file: " + path.name)
            path.chmod(0o600)
            self.conversations[value["id"]] = value
            interrupted = False
            for message in value["messages"]:
                if message.get("status") in ("queued", "running"):
                    message.update(status="error", content="The server restarted before this reply finished. Send your message again to retry.")
                    interrupted = True
            if interrupted:
                value["codex_thread_id"] = None
                self.save(value)
        if not self.conversations:
            self.new_conversation()

    def save(self, conversation):
        conversation["updated_at"] = time.time()
        private_write(self.state / (conversation["id"] + ".json"), json.dumps(conversation, ensure_ascii=False, indent=2) + "\n")

    def ledger_root(self, conversation_id: str | None = None) -> Path:
        if conversation_id:
            if conversation_id in self.overlay_pins:
                return self.overlay_pins[conversation_id]
            if conversation_id in self.overlay_roots:
                return self.overlay_roots[conversation_id]
        return self.root

    def bind_drill_overlay(self, conversation_id: str, fixture_id: str) -> dict:
        if os.environ.get("BOB_WEBCHAT_DRILLS") != "1":
            raise RequestError(403, "Drill overlay is disabled.")
        if not isinstance(fixture_id, str) or not re.fullmatch(r"[0-9a-f-]{36}", fixture_id):
            raise RequestError(400, "Invalid fixture id.")
        base = (HERE / "drill-fixtures").resolve()
        path = (base / fixture_id).resolve()
        if not str(path).startswith(str(base)) or not path.is_dir():
            raise RequestError(404, "Fixture not found.")
        self.get(conversation_id)
        with self.lock:
            self.overlay_roots[conversation_id] = path
        return {"ok": True, "fixture_id": fixture_id}

    def clear_drill_overlay(self, conversation_id: str) -> dict:
        with self.lock:
            self.overlay_roots.pop(conversation_id, None)
        return {"ok": True}

    def record_turn(self, conversation_id=None, **data):
        # The optional handback ledger cannot turn a completed provider reply or
        # a persisted action receipt into a failed/replayed request.
        ask = (data.get("ask") or "").strip()
        if self.ledger_root(conversation_id) == self.root and state_ledger._DRILL_NOISE_RE.search(ask):
            return
        try:
            append_turn(self.ledger_root(conversation_id), **data)
        except OSError:
            if hasattr(self.providers, "circuit"):
                try:
                    self.providers.circuit.log("handback_error", reason="handback write unavailable")
                except OSError:
                    pass

    def new_conversation(self):
        with self.lock:
            value = {"id": str(uuid.uuid4()), "title": "New conversation", "created_at": time.time(),
                     "messages": [], "codex_thread_id": None}
            self.save(value)
            self.conversations[value["id"]] = value
            return self.public(value)

    def public(self, value):
        return {k: copy.deepcopy(v) for k, v in value.items() if k not in ("codex_thread_id", "retry_prompt", "pending_job", "pending_job_id")}

    def get(self, identifier):
        if not isinstance(identifier, str) or not UUID.fullmatch(identifier) or identifier not in self.conversations:
            raise RequestError(404, "Conversation not found.")
        return self.conversations[identifier]

    def history(self, identifier):
        with self.lock:
            return self.public(self.get(identifier))

    def worker_info(self, identifier):
        return self.workers.get(identifier)

    def reply_deadline(self):
        timeouts = getattr(self.providers, "timeouts", {})
        values = list(timeouts.values()) or [180]
        return sum(values[:4]) + 60

    def sweep_stuck(self):
        now = time.time()
        limit = self.reply_deadline()
        with self.lock:
            for identifier, info in list(self.workers.items()):
                if now - info["started_at"] <= limit:
                    continue
                info["cancel"].set()
                value = self.get(identifier)
                if value["messages"] and value["messages"][-1]["status"] in ("queued", "running"):
                    value["messages"][-1].update(status="error",
                        content="This reply timed out. Tap Stop if it is still running, then send again.")
                    value.pop("retry_prompt", None)
                    self.save(value)

    def conversation_list(self):
        with self.lock:
            return {"conversations": [{k: c[k] for k in ("id", "title", "updated_at")}
                                      for c in sorted(self.conversations.values(), key=lambda c: c["updated_at"], reverse=True)],
                    "active_conversation": next(iter(self.workers), None),
                    "active_conversations": list(self.workers)}

    def authenticate(self, cookie):
        try:
            parsed = SimpleCookie(cookie or "")

        except (CookieError, KeyError, ValueError):
            return None
        digest = hashlib.sha256(token.encode()).hexdigest()
        with self.lock:
            session = self.sessions.get(digest)

    def login(self, password):
        with self.lock:
            now = time.time()
            while self.failures and self.failures[0] < now - 60:
                self.failures.popleft()
            if len(self.failures) >= 10:
                raise RequestError(429, "Too many login attempts. Wait one minute and try again.")
            if not isinstance(password, str) or not hmac.compare_digest(password.encode(), self.password.encode()):
                self.failures.append(now)
                raise RequestError(401, "That password did not match. Try again.")
            token = secrets.token_urlsafe(32)
            digest = hashlib.sha256(token.encode()).hexdigest()
            self.sessions = {k: v for k, v in self.sessions.items() if v["expires"] > now}
            if len(self.sessions) >= 64:
                self.sessions.pop(next(iter(self.sessions)))
            self.sessions[digest] = {"expires": now + SESSION_SECONDS, "csrf": secrets.token_urlsafe(32), "digest": digest}
            return token

            raise RequestError(400, "Write a message between 1 and 6,000 characters.")
        if not isinstance(request_id, str) or not UUID.fullmatch(request_id):
            raise RequestError(400, "A valid request ID is required.")
        text = text.strip()
        try:
            text.encode("utf-8")
        except UnicodeError:
            raise RequestError(400, "The message contains invalid text.") from None
        if self.redact(text) != text:
            raise RequestError(400, "This message appears to contain a secret. Remove it before sending; it was not saved or sent to a model.")
        with self.lock:

            for message in conversation["messages"]:

                    if message["content"] != text or message.get("preferred_provider", "auto") != provider:
                        raise RequestError(409, "That request ID was already used for a different message.")
                    return self.public(conversation)
            self.sweep_stuck()
            if identifier in self.workers:
                raise RequestError(409, "Bob is finishing another reply in this conversation. Stop it or wait, then try again.")
            if conversation["messages"] and conversation["messages"][-1]["status"] == "waiting":

    def launch(self, identifier):
        # Caller holds self.lock; the same saved assistant message is reused.
        if not self.closing.is_set():
            cancel = threading.Event()
            if identifier in self.overlay_roots:
                self.overlay_pins[identifier] = self.overlay_roots[identifier]
            worker = threading.Thread(target=self.complete, args=(identifier, cancel), daemon=True)
            self.workers[identifier] = {"thread": worker, "cancel": cancel, "started_at": time.time()}
            worker.start()

    def complete(self, identifier, cancel):
        with self.lock:
            conversation = copy.deepcopy(self.get(identifier))
            value = self.get(identifier)
            if value["messages"] and value["messages"][-1]["status"] == "queued":
                value["messages"][-1].update(status="running", provider=None, attempts=[])
                self.save(value)
        def progress(provider, attempts):
            with self.lock:
                value = self.get(identifier)
                value["messages"][-1].update(status="running", provider=provider, attempts=attempts)
                self.save(value)
        try:
            answer = self.providers.reply(conversation, cancel, progress)
            if cancel.is_set():
                raise Cancelled("Reply stopped.")
            with self.lock:
                value = self.get(identifier)
                value["codex_thread_id"] = answer.pop("codex_thread_id", None)
                value.pop("retry_prompt", None)
                answer["content"] = repair_hollow_progress(self.redact(answer["content"]))
                value["messages"][-1].update(answer, status="done")
                value["messages"][-1].pop("retry_at", None)
                self.save(value)
                user_msg = next((m for m in reversed(value["messages"]) if m["role"] == "user"), {})
                user = user_msg.get("content", "")
                head = draft_head_sha(self.root)
                commits = [head] if head else []
                self.record_turn(conversation_id=identifier, ask=user, provider=answer.get("provider"),
                            jobs=[], files=[], commits=commits, confirmations=[], refusals=[],
                            open_items=[user[:200]] if user else [],
                            request_id=user_msg.get("request_id"))
                try:
                    ledger = self.ledger_root(identifier)
                    if not (ledger == self.root and state_ledger._DRILL_NOISE_RE.search(user or "")):
                        state_ledger.record_webchat_turn(
                            ledger, ask=user, provider=answer.get("provider"), request_id=user_msg.get("request_id"),
                        )
                except OSError:
                    pass
        except AllDown as exc:
            with self.lock:
                value = self.get(identifier)
                value["codex_thread_id"] = None
                value["retry_prompt"] = exc.prompt
                value["messages"][-1].update(status="cancelled" if cancel.is_set() else "waiting",
                    provider=None, content="Reply stopped." if cancel.is_set() else str(exc),
                    attempts=exc.attempts, retry_at=exc.retry_at)
                self.save(value)
        except Exception as exc:
            with self.lock:
                value = self.get(identifier)
                value["codex_thread_id"] = None
                value.pop("retry_prompt", None)
                message = str(exc) if isinstance(exc, ProviderError) else "The reply failed unexpectedly. Your message is saved; please try again."
                value["messages"][-1].update(status="cancelled" if isinstance(exc, Cancelled) else "error", content=self.redact(message))
                self.save(value)
        finally:
            with self.lock:
                self.workers.pop(identifier, None)
                self.overlay_pins.pop(identifier, None)

    def stop(self, identifier):
        with self.lock:
            value = self.get(identifier)
            info = self.workers.get(identifier)
            if info:
                info["cancel"].set()
            elif value["messages"] and value["messages"][-1]["status"] == "waiting":
                value["messages"][-1].update(status="cancelled", content="Queued request stopped.")
                value.pop("retry_prompt", None)
                self.save(value)
                if hasattr(self.providers, "circuit"):
                    self.providers.circuit.log("queue_cancelled", conversation_id=identifier)

    def retry_queued(self):
        with self.lock:
            self.sweep_stuck()
            queued = sorted((c for c in self.conversations.values() if c["messages"] and c["messages"][-1]["status"] == "waiting"), key=lambda c: c["messages"][-1]["created_at"])
            for value in queued:
                if value["id"] in self.workers:
                    continue
                message = value["messages"][-1]
                if message.get("retry_at", 0) <= time.time():
                    message.update(status="queued", content="", provider=None)
                    self.save(value)
                    if hasattr(self.providers, "circuit"):
                        self.providers.circuit.log("queue_retry", conversation_id=value["id"])
                    self.launch(value["id"])
                    return

    def start_background(self):
        self.providers.start_probes()
        def loop():
            while not self.closing.wait(1):
                try:
                    if not self.closing.is_set():
                        self.retry_queued()
                        self.sweep_stuck()
                except Exception:
                    self.providers.circuit.log("queue_error", reason="queued retry could not start")
        self.queue_thread = threading.Thread(target=loop, daemon=True)
        self.queue_thread.start()

    def close(self):
        self.closing.set()
        with self.lock:
            for info in self.workers.values():
                info["cancel"].set()
        if self.queue_thread:
            self.queue_thread.join(timeout=3)
        for info in list(self.workers.values()):
            info["thread"].join(timeout=10)
        if hasattr(self.providers, "close"):
            self.providers.close()

    @staticmethod
    def _launchd_job_loaded(label: str) -> bool | None:
        if sys.platform != "darwin":
            return None
        try:
            proc = subprocess.run(
                ["/bin/launchctl", "print", f"gui/{os.getuid()}/{label}"],
                capture_output=True,
                timeout=3,
                check=False,
            )
            return proc.returncode == 0
        except (OSError, subprocess.TimeoutExpired):
            return None

    def _cached_launchd_loaded(self, label: str) -> bool | None:
        cache = getattr(self, "_launchd_probe_cache", {})
        now = time.time()
        hit = cache.get(label)
        if hit and now - hit[1] < 60:
            return hit[0]
        value = self._launchd_job_loaded(label)
        cache[label] = (value, now)
        self._launchd_probe_cache = cache
        return value

    def _self_improve_schedule_hint(self, plist: Path) -> str | None:
        if not plist.is_file():
            return None
        try:
            text = plist.read_text(errors="replace")
        except OSError:
            return None
        hour = minute = None
        for key, var in (("Hour", "hour"), ("Minute", "minute")):
            match = re.search(rf"<key>{key}</key>\s*<integer>(\d+)</integer>", text)
            if match:
                if var == "hour":
                    hour = int(match.group(1))
                else:
                    minute = int(match.group(1))
        if hour is None:
            return None
        minute = minute if minute is not None else 0
        return f"Scheduled daily ~{hour:02d}:{minute:02d} CT on the mini (launchd)."

    def round_changelog(self) -> dict:
        import re

        def round_num(path: Path) -> int:
            match = re.search(r"R(\d+)", path.name)
            return int(match.group(1)) if match else 0

        paths = sorted(
            self.root.glob("conductor/reviews/WEBCHAT-READY-R*-RESULT.md"),
            key=round_num,
            reverse=True,
        )
        if not paths:
            return {}
        path = paths[0]
        name = path.name
        round_no = round_num(path) or None
        line = ""
        try:
            text = path.read_text(errors="replace")
            section = re.search(r"^## Changelog\s*\n+(.+?)(?:\n## |\Z)", text, re.M | re.S)
            if section:
                for row in section.group(1).splitlines():
                    cleaned = row.strip()
                    if cleaned and not cleaned.startswith("#"):
                        line = cleaned[:220]
                        break
            if not line:
                explicit = re.search(r"^\*\*Changelog:\*\*\s*(.+)$", text, re.M)
                if explicit:
                    line = explicit.group(1).strip()[:220]
            if not line:
                for row in text.splitlines():
                    if "Time-to-value" in row or "visible improvement" in row.lower():
                        line = row.strip().lstrip("|").rstrip("|").strip()[:220]
                        break
            if not line:
                for row in text.splitlines():
                    if row.startswith("|") and "iPhone" in row:
                        line = row.strip().lstrip("|").rstrip("|").strip()[:220]
                        break
        except OSError:
            pass
        return {"round": round_no, "file": name, "line": line or "See latest round result on the mini."}

class Server(ThreadingHTTPServer):
    daemon_threads = True
    allow_reuse_address = True

    def __init__(self, address, chat, public_origin=""):
        if address[0] != HOST:
            raise ValueError("Bob web chat must bind to 127.0.0.1")
        self.chat = chat
        super().__init__(address, Handler)
        port = self.server_address[1]
        self.origins = {f"127.0.0.1:{port}": f"http://127.0.0.1:{port}", f"localhost:{port}": f"http://localhost:{port}"}
        if public_origin:
            parsed = urlsplit(public_origin)
            if (parsed.scheme != "https" or not parsed.hostname or not parsed.hostname.endswith(".ts.net")
                    or parsed.path not in ("", "/") or parsed.query or parsed.fragment or parsed.username or parsed.password):
                self.server_close()
                raise ValueError(".public-origin must be an exact https:// device.ts.net origin")
            self.origins[parsed.netloc.lower()] = public_origin.rstrip("/")


class Handler(BaseHTTPRequestHandler):
    server_version = "BobWebchat/1"

    def setup(self):
        super().setup()
        self.connection.settimeout(10)

    def log_message(self, *_args):
        pass  # No URLs, cookies, prompts, passwords or provider output in HTTP logs.

    def send(self, status, data, content_type="application/json; charset=utf-8", headers=None):
        if isinstance(data, (dict, list)):
            data = json.dumps(data, ensure_ascii=False)
        if isinstance(data, str):
            data = data.encode()
        self.send_response(status)
        extra = {"X-Bob-Webchat-Drills": "1"} if os.environ.get("BOB_WEBCHAT_DRILLS") == "1" else {}
        for key, value in {
            "Content-Type": content_type, "Content-Length": str(len(data)), "Cache-Control": "no-store",
            "X-Content-Type-Options": "nosniff", "X-Frame-Options": "DENY", "Referrer-Policy": "no-referrer",
            "Content-Security-Policy": "default-src 'none'; script-src 'self'; style-src 'self'; img-src 'self'; media-src 'self'; connect-src 'self'; form-action 'self'; frame-ancestors 'none'; base-uri 'none'",
            "X-Bob-Webchat": "1", **extra, **(headers or {}),
        }.items():
            self.send_header(key, value)
        self.end_headers()
        self.wfile.write(data)

    def body(self, form=False, limit=None):
        if self.headers.get("Transfer-Encoding"):
            raise RequestError(400, "Transfer encoding is not supported.")
        cap = limit or MAX_BODY
        try:
            size = int(self.headers.get("Content-Length", "0"))
        except ValueError:
            raise RequestError(400, "Invalid content length.") from None
        if not 0 < size <= cap:
            raise RequestError(413, "Request is empty or too large.")
        mime = self.headers.get("Content-Type", "").split(";", 1)[0].strip().lower()
        if mime != ("application/x-www-form-urlencoded" if form else "application/json"):
            raise RequestError(415, "Unsupported content type.")
        try:
            raw = self.rfile.read(size).decode("utf-8")
            data = {k: v[0] for k, v in parse_qs(raw).items()} if form else json.loads(raw)
            if not isinstance(data, dict):
                raise ValueError()
            return data
        except (UnicodeError, ValueError):
            raise RequestError(400, "Invalid request body.") from None

    def cookie(self, token, clear=False):
        secure = self.origin.startswith("https://")
        return f"{COOKIE}={token}; Path=/; HttpOnly; SameSite=Strict; Max-Age={0 if clear else SESSION_SECONDS}" + ("; Secure" if secure else "")

    def dispatch(self):
        try:
            hosts = self.headers.get_all("Host", [])
            if len(hosts) != 1 or hosts[0].lower() not in self.server.origins:
                raise RequestError(403, "Unrecognized host.")
            self.origin = self.server.origins[hosts[0].lower()]
            self.path_only = urlsplit(self.path).path
            session = self.server.chat.authenticate(self.headers.get("Cookie"))
            if self.path_only == "/api" or self.path_only.startswith("/api/"):
                if not session:
                    raise RequestError(401, "Please log in.")
            if self.command == "POST":
                origins = self.headers.get_all("Origin", [])
                if (origins and origins != [self.origin]) or self.headers.get("Sec-Fetch-Site") in ("cross-site", "same-site"):
                    raise RequestError(403, "Cross-site requests are not allowed.")
                if self.path_only.startswith("/api/"):
                    csrf = self.headers.get("X-CSRF-Token", "")
                    if not hmac.compare_digest(csrf.encode(), session["csrf"].encode()):
                        raise RequestError(403, "Invalid request token. Reload the page.")
                self.post(session)
            elif self.command == "GET":
                self.get(session)
            else:
                raise RequestError(405, "Method not allowed.")
        except RequestError as exc:
            self.send(exc.status, {"error": exc.message})
        except (BrokenPipeError, ConnectionResetError, TimeoutError):
            pass
        except Exception:
            self.send(500, {"error": "The request failed. Please try again."})

    do_GET = dispatch
    do_POST = dispatch
    do_OPTIONS = dispatch
    do_PUT = dispatch
    do_DELETE = dispatch
    do_PATCH = dispatch

    def stream_alignment(self, chat):
        try:
            import alignment_events as align_ev  # noqa: WPS433
        except ImportError:
            raise RequestError(503, "Alignment stream unavailable.") from None
        query = parse_qs(urlsplit(self.path).query)
        try:
            since = int(query.get("since", ["0"])[0] or 0)
        except ValueError:
            raise RequestError(400, "Invalid since parameter.") from None
        self.connection.settimeout(45.0)
        self.send_response(200)
        for key, value in {
            "Content-Type": "text/event-stream; charset=utf-8",
            "Cache-Control": "no-store",
            "X-Content-Type-Options": "nosniff",
            "X-Bob-Webchat": "1",
        }.items():
            self.send_header(key, value)
        self.end_headers()
        deadline = time.time() + 40
        cursor = since
        try:
            while time.time() < deadline:
                batch = align_ev.read_events(chat.root, since_seq=cursor)
                for row in batch:
                    cursor = max(cursor, int(row.get("seq") or 0))
                    payload = json.dumps(row, ensure_ascii=False).encode()
                    self.wfile.write(b"data: " + payload + b"\n\n")
                self.wfile.flush()
                if batch:
                    self.wfile.write(f"event: ping\ndata: {{\"latest_seq\":{cursor}}}\n\n".encode())
                    self.wfile.flush()
                time.sleep(2)
        except (BrokenPipeError, ConnectionResetError, TimeoutError):
            pass

                raise RequestError(403, "Drill overlay is disabled.")
            data = self.body()
            return self.send(200, chat.bind_drill_overlay(data.get("conversation_id"), data.get("fixture_id")))
        if route == "/api/drill/overlay/clear":
            if os.environ.get("BOB_WEBCHAT_DRILLS") != "1":
                raise RequestError(403, "Drill overlay is disabled.")
            data = self.body()
            chat.get(data.get("conversation_id"))
            return self.send(200, chat.clear_drill_overlay(data.get("conversation_id")))
        if route == "/api/audio/upload":
            data = self.body(limit=MAX_AUDIO_BODY)
            name = data.get("filename") or "upload.mp3"
            raw = data.get("data_base64")
            if not isinstance(raw, str):
                raise RequestError(400, "Audio payload required.")
            try:
                blob = base64.b64decode(raw, validate=True)
            except ValueError:
                raise RequestError(400, "Invalid audio encoding.") from None
            suffix = Path(name).suffix or ".mp3"
            try:
                stored = store_upload(chat.root, blob, suffix)
            except ValueError as exc:
                raise RequestError(400, str(exc)) from exc
            chat.record_turn(ask=f"audio upload {name}", provider=None, jobs=[], files=[stored["path"]],
                            commits=[], confirmations=[], refusals=[], open_items=[])
            return self.send(201, stored)
        raise RequestError(404, "Not found.")


def main():
    os.umask(0o077)
    with (HERE / ".server.lock").open("a") as lock:
        try:
            fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            raise SystemExit("Bob web chat is already running.")
        chat = Chat()
        origin_path = HERE / ".public-origin"
        origin = origin_path.read_text().strip() if origin_path.exists() else ""
        server = Server((HOST, PORT), chat, origin)
        chat.start_background()
        def shutdown(_signum, _frame):
            with chat.lock:
                for info in chat.workers.values():
                    info["cancel"].set()
            threading.Thread(target=server.shutdown, daemon=True).start()
        for signum in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP):
            signal.signal(signum, shutdown)
        print(f"Bob web chat listening on http://{HOST}:{PORT} (auth via local session file; secret never logged)", flush=True)
        try:
            server.serve_forever(poll_interval=0.25)
        finally:
            server.server_close()
            chat.close()


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Persistent song projects for Web Bob (lyrics, chords, notes, versions)."""
from __future__ import annotations

import json
import os
import re
import secrets
import time
import uuid
from pathlib import Path

PROJECT_ID = re.compile(r"^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$", re.I)

FOLDER = Path("conductor/webchat/song-projects")
INDEX = "index.json"
MAX_BODY = 120_000


def _dir(root: Path) -> Path:
    path = root / FOLDER
    path.mkdir(parents=True, exist_ok=True)
    return path


def _write(path: Path, value: str) -> None:
    temporary = path.with_name("." + path.name + "." + secrets.token_hex(6))
    try:
        with open(os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), "w") as handle:
            handle.write(value)
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(temporary, path)
    finally:
        temporary.unlink(missing_ok=True)


def _load_index(root: Path) -> dict:
    path = _dir(root) / INDEX
    if not path.exists():
        return {"projects": []}
    return json.loads(path.read_text())


def _save_index(root: Path, data: dict) -> None:
    _write(_dir(root) / INDEX, json.dumps(data, ensure_ascii=False, indent=2) + "\n")


def list_projects(root: Path) -> list[dict]:
    data = _load_index(root)
    rows = sorted(data.get("projects", []), key=lambda row: row.get("updated_at", 0), reverse=True)
    return [{k: row[k] for k in ("id", "title", "updated_at", "created_at") if k in row} for row in rows]


def get_project(root: Path, project_id: str) -> dict:
    if not PROJECT_ID.fullmatch(project_id):
        raise FileNotFoundError(project_id)
    path = _dir(root) / f"{project_id}.json"
    if not path.exists():
        raise FileNotFoundError(project_id)
    return json.loads(path.read_text())


def save_project(root: Path, payload: dict) -> dict:
    title = (payload.get("title") or "Untitled song").strip()[:120]
    lyrics = (payload.get("lyrics") or "")[:MAX_BODY]
    chords = (payload.get("chords") or "")[:MAX_BODY]
    notes = (payload.get("notes") or "")[:MAX_BODY]
    project_id = payload.get("id")
    now = time.time()
    if project_id and not PROJECT_ID.fullmatch(project_id):
        raise ValueError("invalid project id")
    if len(lyrics.strip()) + len(chords.strip()) < 3 and not project_id:
        raise ValueError("Add a few characters of lyrics or chords before saving.")
    if project_id:
        project = get_project(root, project_id)
        project.update(title=title, lyrics=lyrics, chords=chords, notes=notes, updated_at=now)
    else:
        project_id = str(uuid.uuid4())
        project = {"id": project_id, "title": title, "lyrics": lyrics, "chords": chords, "notes": notes,
                   "created_at": now, "updated_at": now, "versions": []}
    if payload.get("version_note"):
        project.setdefault("versions", []).append(
            {"at": now, "note": str(payload["version_note"])[:500], "lyrics": lyrics[:8000]})
        project["versions"] = project["versions"][-30:]
    _write(_dir(root) / f"{project_id}.json", json.dumps(project, ensure_ascii=False, indent=2) + "\n")
    data = _load_index(root)
    rows = [row for row in data.get("projects", []) if row.get("id") != project_id]
    rows.append({"id": project_id, "title": title, "created_at": project["created_at"], "updated_at": now})
    _save_index(root, {"projects": rows})
    return project


def snapshot_for_prompt(root: Path, limit: int = 6000) -> str:
    lines = []
    for row in list_projects(root)[:8]:
        try:
            project = get_project(root, row["id"])
            snippet = (project.get("lyrics") or "").strip().splitlines()
            preview = "\n".join(snippet[:12])
            chords = (project.get("chords") or "").strip().splitlines()
            chord_line = "\n".join(chords[:4])
            notes = (project.get("notes") or "").strip()[:400]
            block = f"### {project['title']} ({project['id'][:8]})\n{preview}"
            if chord_line:
                block += f"\nChords: {chord_line}"
            if notes:
                block += f"\nNotes: {notes}"
            lines.append(block)
        except (OSError, json.JSONDecodeError, KeyError):
            lines.append(f"### (corrupt project {row.get('id', '?')[:8]}) — fix in Song panel")
            continue
    text = "\n\n".join(lines)
    return text[:limit]


_MUSIC_COACH_RE = re.compile(
    r"\b(?:(?:my |a )?song|lyrics|chorus|verse|bridge|hook|chords?|progression|melody|arrange|rhyme|meter|tempo|bpm|structure|time signature|4/4|3/4|6/8|modulation|key change|outro|fade|ending|pre[- ]?chorus|capo|riffs?|vamp|dynamics?|breakdown|build[- ]?up|drop)\b",
    re.I,
)


def try_music_coach_reply(text: str) -> str | None:
    """Cheap songwriting partner tips (no API spend)."""
    stripped = (text or "").strip()
    if not stripped or len(stripped) > 280 or not _MUSIC_COACH_RE.search(stripped):
        return None
    lower = stripped.lower()
    if "modulation" in lower or "key change" in lower:
        return (
            "Modulation: move up a half-step or whole-step for the final chorus lift (common in pop). "
            "Set up with a dominant chord (V) of the new key one bar before the change — e.g. G7 before C major. "
            "Note old and new keys in Songs → notes so versions and Bob stay aligned."
        )
    music_ctx = re.search(
        r"\b(song|lyrics|chorus|verse|guitar|track|riff|hook|arrange)\b",
        lower,
    )
    if re.search(r"\bintro\b", lower) and music_ctx:
        return (
            "Intro: one riff or motif that foreshadows the chorus hook — keep it 4–8 bars, no lyrics required. "
            "Drop the energy after bar 4 so verse 1 feels like a breath. "
            "Note the intro idea in Songs → notes; Logic markers are confirm-first."
        )
    if "pre-chorus" in lower or "prechorus" in lower or "pre chorus" in lower:
        return (
            "Pre-chorus: build tension with rising chords (e.g. vi–IV–V) and a shorter melodic line than the verse. "
            "Land on the dominant (V) right before the chorus downbeat. "
            "Save lyric tweaks as a new Songs version so Bob sees the lift."
        )
    if "capo" in lower:
        return (
            "Capo: match the singer’s comfortable range — write shapes in friendly keys (G, C, D) and note capo fret in Songs → chords. "
            "Same chord names on paper; the sounding key shifts up. "
            "Record which fret you used in notes for the handback."
        )
    if re.search(r"\b(dynamic|breakdown|build[- ]?up|drop)\b", lower):
        if not music_ctx:
            return None
        return (
            "Dynamics: verse stays lean (guitar + vocal), pre-chorus adds bass or pad, chorus full band. "
            "Breakdown = strip to one instrument for 4–8 bars before the final lift. "
            "Note the map in Songs → notes; Logic level automation is confirm-first."
        )
    if "double chorus" in lower or "post-chorus" in lower or "post chorus" in lower:
        return (
            "Double chorus: play the full chorus twice back-to-back; lift drums or harmony on the second pass. "
            "A post-chorus can be instrumental (hook riff only) between verse and chorus. "
            "Note the map in Songs → notes; Logic arrangement is confirm-first."
        )
    if "tag line" in lower or "tagline" in lower or ("chorus" in lower and "tag" in lower):
        return (
            "Tag line: repeat one hook phrase after the chorus (often the title) — 2–4 words max, same melody as the chorus hook. "
            "Use it on the outro vamp or double-chorus ending. "
            "Save the tag lyric as a new Songs version so Bob sees the final hook."
        )
    if "riff" in lower or "vamp" in lower:
        return (
            "Riff/vamp: repeat a 1–2 bar figure; change one note on the 4th repeat for motion. "
            "Use it under the chorus tag or as an outro fade. "
            "Hum it into Songs audio upload if you want a reference take ($0 local sketch later)."
        )
    if "outro" in lower or "ending" in lower or "fade" in lower:
        return (
            "Outro: repeat the chorus hook once, strip instruments for the last line, or tag a new lyric over the vamp. "
            "A fade works on sketch demos; in Logic, mark the outro region confirm-first before bounce. "
            "Save the lyric tweak as a new Songs version."
        )
    if "chord" in lower or "progression" in lower:
        return (
            "For chords, try a simple loop (verse: I–V–vi–IV, chorus lift to IV or vi). "
            "Match the mood in the Songs tab — save a version after each pass. "
            "Local sketch demo ($0) can render a rough MP3 from lyrics + style when you are ready."
        )
    if "chorus" in lower or "hook" in lower:
        return (
            "Chorus tip: one clear emotional line, repeat the title, keep lines short for singing. "
            "Contrast with the verse (higher energy or wider melody). "
            "Drop a draft in Songs → lyrics, then ask me to tighten a specific line."
        )
    if "rhyme" in lower or "meter" in lower:
        return (
            "Rhyme without forcing: pick one rhyme sound per section (e.g. -ay) and leave one line unrhymed for breath. "
            "For meter, read lines aloud and trim syllables until stresses land on beats 1 and 3 in 4/4. "
            "Save a version in Songs after each rewrite so Bob sees the history."
        )
    if "melody" in lower or "arrange" in lower:
        return (
            "Melody: start with the chorus on fewer notes than the verse; leave space at phrase ends. "
            "Arrangement: add one layer at a time (guitar → bass → pad). "
            "Use Songs for structure notes; Logic on the mini is confirm-first for tempo/markers/bounce."
        )
    if "bridge" in lower or "structure" in lower or "verse" in lower:
        return (
            "Structure: verse sets the story, chorus is the emotional headline, bridge is one new angle before the final chorus. "
            "Try 8-bar sections in 4/4; label them in Songs → notes so versions stay clear. "
            "When a section feels long, cut a line instead of adding chords."
        )
    if "time signature" in lower or "4/4" in lower or "3/4" in lower or "6/8" in lower:
        return (

        )
    if "tempo" in lower or "bpm" in lower or ("key" in lower and "change" not in lower):
        return (

    return (

"""Dashboard live_ops + conductor read snapshot for Web Bob status panel."""
from __future__ import annotations

import json
import re
import subprocess
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

LIVE_OPS_MAX_AGE = 120.0
_REFRESH_COOLDOWN = 45.0
_last_refresh_spawn = 0.0


def maybe_spawn_dashboard_refresh(root: Path) -> bool:
    """Non-blocking dashboard-updater tick when dashboard.json is stale."""
    global _last_refresh_spawn
    now = time.time()
    if now - _last_refresh_spawn < _REFRESH_COOLDOWN:
        return False
    dash = root / "conductor/dashboard.json"
    if dash.exists() and now - dash.stat().st_mtime <= LIVE_OPS_MAX_AGE:
        return False
    lock = root / "conductor/locks/webchat-dashboard-refresh.lock"
    lock.parent.mkdir(parents=True, exist_ok=True)
    try:
        handle = lock.open("a")
        try:
            import fcntl

            fcntl.flock(handle.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        except (OSError, ImportError):
            handle.close()
            return False
        handle.close()
    except OSError:
        return False
    services = root / "conductor/services"
    script = (
        f"import sys; sys.path.insert(0, '{services}'); "
        "from runtime import dashboard_cycle; dashboard_cycle()"
    )
    try:
        subprocess.Popen(
            [sys.executable, "-c", script],
            cwd=str(root),
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            start_new_session=True,
        )
        _last_refresh_spawn = now
        return True
    except OSError:
        return False


def draft_head_sha(root: Path) -> str | None:
    conductor = root / "conductor"
    for base in (conductor, root):
        if not (base / ".git").exists():
            continue
        try:
            return subprocess.check_output(
                ["git", "-C", str(base), "rev-parse", "--short", "HEAD"],
                text=True,
                timeout=5,
            ).strip()
        except (subprocess.SubprocessError, OSError):
            continue
    return None


def _parse_generated_at(live_ops: dict) -> float | None:
    raw = live_ops.get("generated_at")
    if not raw:
        return None
    try:
        if raw.endswith("Z"):
            raw = raw[:-1] + "+00:00"
        return datetime.fromisoformat(raw).timestamp()
    except (TypeError, ValueError):
        return None


def inline_live_ops(root: Path) -> dict:
    services = str(root / "conductor/services")
    if services not in sys.path:
        sys.path.insert(0, services)
    from runtime import snapshot

    return snapshot().get("live_ops") or {}


def load_live_ops(root: Path, *, max_age: float = LIVE_OPS_MAX_AGE) -> dict:
    dash_path = root / "conductor/dashboard.json"
    meta = {"source": "none", "dashboard_modified_at": None, "generated_at": None, "stale": True}
    live_ops: dict = {}
    if dash_path.exists():
        try:
            meta["dashboard_modified_at"] = dash_path.stat().st_mtime
            payload = json.loads(dash_path.read_text(errors="replace"))
            live_ops = payload.get("live_ops") or {}
            meta["generated_at"] = live_ops.get("generated_at")
            generated = _parse_generated_at(live_ops)
            age = time.time() - generated if generated else time.time() - meta["dashboard_modified_at"]
            if live_ops and age <= max_age:
                meta.update(source="dashboard.json", stale=False)
                return {"live_ops": live_ops, "meta": meta}
            meta["stale"] = True
            maybe_spawn_dashboard_refresh(root)
        except (OSError, json.JSONDecodeError, ValueError):
            live_ops = {}
    try:
        live_ops = inline_live_ops(root)
        meta.update(source="inline_snapshot", stale=False, generated_at=live_ops.get("generated_at"))
    except Exception:
        meta["source"] = "dashboard_stale" if live_ops else "unavailable"
        meta["stale"] = True
    return {"live_ops": live_ops, "meta": meta}
