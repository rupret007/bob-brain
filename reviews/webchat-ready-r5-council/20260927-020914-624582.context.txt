#!/usr/bin/env python3
"""Plain-English live ops slice for dashboard.json and agents-status.json (no secrets)."""
from __future__ import annotations

import json
import re
import time
from pathlib import Path

from datetime import datetime, timezone

from state import CONDUCTOR, ROOT

MINIMAX_DISPLAY_CAP_USD = 59.0
_SCORE_ROW = re.compile(r"^\|\s*([^|]+?)\s*\|\s*[^|]*\|\s*\*\*([^*]+?)\*\*", re.M)
_HOME = re.compile(r"/Users/[^\s\"']+")


def _redact_line(line: str) -> str:
    line = _HOME.sub("[mini]", line)
    for pattern in (r"gh[pousr]_[A-Za-z0-9]{20,}", r"AIza[0-9A-Za-z_-]{20,}", r"sk-[A-Za-z0-9_-]{16,}"):
        line = re.sub(pattern, "[redacted]", line)
    return line.strip()[:240]


def _tail_lines(path: Path, count: int = 3) -> list[str]:
    if not path.is_file():
        return []
    try:
        text = path.read_text(errors="replace")
    except OSError:
        return []
    lines = [_redact_line(line) for line in text.splitlines() if line.strip()]
    return lines[-count:]


def _job_log_lines(job_dir: Path, job_id: str) -> list[str]:
    candidates = [
        job_dir / "worker.log",
        job_dir / "run.log",
        job_dir / "screen.log",
        ROOT / "logs" / (job_id + ".log"),
        ROOT / "logs" / (job_id + "-screen.log"),
    ]
    for path in candidates:
        lines = _tail_lines(path)
        if lines:
            return lines
    return []


def _plain_english(job_id: str, engine: str | None, state: str, log_lines: list[str]) -> str:
    if log_lines:
        return log_lines[-1]
    if state == "running":
        return f"{engine or 'worker'} job {job_id[:24]} is active on a screen session."
    if state == "blocked":
        return f"{job_id[:24]} is blocked waiting on recovery."
    return f"{job_id[:24]} is {state}."


def _git_head(job_dir: Path) -> dict:
    if not job_dir.is_dir():
        return {}
    git_dir = job_dir / ".git"
    head = job_dir / ".git" / "HEAD"
    if not head.is_file() and not git_dir.is_file():
        return {}
    from command import run

    result = run(["git", "-C", str(job_dir), "log", "-1", "--format=%h|%s"], capture_output=True, text=True, timeout=8)
    if result.returncode or "|" not in result.stdout:
        return {}
    sha, message = result.stdout.strip().split("|", 1)
    return {"sha": sha[:12], "message": _redact_line(message)[:160]}


def _webchat_round() -> int | None:
    reviews = CONDUCTOR / "reviews"
    best = 0
    for path in reviews.glob("WEBCHAT-READY-R*-RESULT.md"):
        match = re.search(r"R(\d+)-RESULT", path.name)
        if match:
            best = max(best, int(match.group(1)))
    return best or None


def _scores_table() -> list[dict]:
    reviews = sorted((CONDUCTOR / "reviews").glob("WEBCHAT-READY-R*-RESULT.md"),
                     key=lambda p: p.stat().st_mtime, reverse=True)
    if not reviews:
        return []
    text = reviews[0].read_text(errors="replace")
    rows = []
    for area, score in _SCORE_ROW.findall(text):
        area = area.strip()
        score = score.strip().split()[0]
        if area.lower() in ("area", "check"):
            continue
        if re.match(r"^[\d.]+$", score.split("·")[0].strip()):
            rows.append({"area": area, "score": score.split("·")[0].strip()})
    return rows[:16]


def _lane_roles(lanes: dict, sessions: set[str]) -> dict:
    building = reviewing = None
    for name, lane in (lanes or {}).items():
        state = (lane or {}).get("state", "")
        if state == "busy" and name in ("cursor", "codex"):
            building = name
        if state == "busy" and name in ("gemini", "minimax", "claude"):
            reviewing = reviewing or name
    if "svc-reviewer" in sessions:
        reviewing = reviewing or "reviewer"
    if "svc-job-runner" in sessions:
        building = building or "scheduler"
    return {"building": building, "reviewing": reviewing}


def _stalled(record: dict, now: float, live: bool) -> str:
    state = record.get("state", "unknown")
    if state in ("harvested", "blocked"):
        return "finished" if state == "harvested" else "stalled"
    launched = float(record.get("launched_at") or record.get("timestamp") or now)
    age = now - launched
    if not live and state == "running":
        return "stalled"
    if live and age > 6 * 3600:
        return "stalled"
    return "running" if live else state


def build_live_ops(*, sessions: dict, jobs_map: dict, lanes: dict, spend: dict, now: float | None = None) -> dict:
    now = time.time() if now is None else now
    session_names = set(sessions)
    svc_screens = {name for name in session_names if name.startswith("svc-") or name.startswith("drill-")}
    active_jobs = []
    for job_id, record in jobs_map.items():
        if record.get("state") not in ("running", "launched", "queued"):
            continue
        live = job_id in session_names
        if not live and record.get("state") != "running":
            continue
        job_dir = Path(record.get("job_dir") or ROOT / job_id)
        log_lines = _job_log_lines(job_dir, job_id)
        launched = float(record.get("launched_at") or record.get("timestamp") or now)
        active_jobs.append({
            "job_id": job_id,
            "engine": record.get("engine") or record.get("model"),
            "state": record.get("state"),
            "live_screen": live,
            "stall_state": _stalled(record, now, live),
            "started_at": datetime.fromtimestamp(launched, tz=timezone.utc).strftime("%Y-%m-%dT%H:%M:%S+00:00") if launched else None,
            "elapsed_seconds": round(max(0, now - launched)),
            "plain_now": _plain_english(job_id, record.get("engine"), record.get("state", ""), log_lines),
            "log_tail": log_lines,
            "last_commit": _git_head(job_dir),
            "round": _webchat_round(),
        })
    active_jobs.sort(key=lambda row: (0 if row["live_screen"] else 1, -row["elapsed_seconds"]))
    minimax = spend.get("minimax") or {}
    roles = _lane_roles(lanes, session_names)
    return {
        "generated_at": datetime.fromtimestamp(now, tz=timezone.utc).strftime("%Y-%m-%dT%H:%M:%S+00:00"),
        "webchat_round": _webchat_round(),
        "scores": _scores_table(),
        "minimax_spend": {
            "committed_usd": round(float(minimax.get("committed_usd") or 0), 4),
            "cap_usd": MINIMAX_DISPLAY_CAP_USD,
            "remaining_usd": round(max(0, MINIMAX_DISPLAY_CAP_USD - float(minimax.get("committed_usd") or 0)), 4),
        },
        "roles": roles,
        "service_screens": sorted(svc_screens),
        "running_jobs": active_jobs[:12],
    }


def public_live_ops(live_ops: dict) -> dict:
    """Strip local paths and ids for GitHub Pages payload."""
    jobs = []
    for row in live_ops.get("running_jobs") or []:
        jobs.append({
            "engine": row.get("engine") or "worker",
            "stall_state": row.get("stall_state"),
            "elapsed_seconds": row.get("elapsed_seconds"),
            "plain_now": row.get("plain_now"),
            "log_tail": row.get("log_tail"),
            "last_commit": row.get("last_commit"),
            "round": row.get("round"),
            "live_screen": row.get("live_screen"),
        })
    return {
        "webchat_round": live_ops.get("webchat_round"),
        "scores": live_ops.get("scores"),
        "minimax_spend": live_ops.get("minimax_spend"),
        "roles": live_ops.get("roles"),
        "running_jobs": jobs,
    }

#!/usr/bin/env python3
"""Bob ↔ Web Bob handback live drill (run twice). Offline ledger checks + live takeover ask."""
from __future__ import annotations

import sys
import time
import uuid
from pathlib import Path

from handback import append_turn, refresh_summary

HANDBACK_FILES = ("BOB-STATE.md", "HANDBACK-LOG.md", "HANDBACK-SUMMARY.md", ".bob-last-seen")


def backup_handback(root: Path) -> dict[str, bytes]:
    folder = root / "conductor/handback"
    saved: dict[str, bytes] = {}
    for name in HANDBACK_FILES:
        path = folder / name
        if path.exists():
            saved[name] = path.read_bytes()
    return saved


def restore_handback(root: Path, saved: dict[str, bytes]) -> None:
    folder = root / "conductor/handback"
    folder.mkdir(parents=True, exist_ok=True)
    for name in HANDBACK_FILES:
        path = folder / name
        if name in saved:
            path.write_bytes(saved[name])
        elif path.exists():
            path.unlink()


def ledger_round(root: Path, tag: str) -> dict:

def live_round(url: str, tag: str) -> dict:
    parsed_host = url.split("://", 1)[-1].rstrip("/")
    client = Curl(url if "://" in url else "http://" + parsed_host)
    conv = client.new()
    started = time.monotonic()
    prompts = [
        "From BOB-STATE and handback only: what's running right now? Include the exact screen/job id.",
        f"From BOB-STATE only: quote the Running line for handback drill `{tag}` in one sentence.",
    ]
    last = {}
    for prompt in prompts:
        client.submit(conv, prompt, str(uuid.uuid4()))
        value = client.wait(conv)
        msg = value["messages"][-1]
        body = (msg.get("content") or "").lower()
        tag_bits = tag.replace("-", " ").split()
        specific = f"drill-{tag}" in body or f"handback drill {tag}" in body or all(bit in body for bit in tag_bits)
        mentions_running = specific and any(word in body for word in ("running", "drill", "screen", "active"))
        last = {
            "tag": tag,
            "status": msg.get("status"),
            "provider": msg.get("provider"),
            "elapsed_seconds": round(time.monotonic() - started, 3),
            "mentions_running": mentions_running,
            "snippet": (msg.get("content") or "")[:220],
            "live_ok": msg.get("status") == "done" and mentions_running,
        }
        if last["live_ok"]:
            break
    client.close()
    return last

def hard_rule_refusal(text):
    if re.search(r"\b(?:merge\s+(?:(?:pr|pull request)\s*)?#?\d+|undraft|force[ -]push|gh\s+pr\s+(?:merge|ready))\b", text, re.I):
        return "I can review the draft, but I won't merge, undraft, or force-push. Drafts only; no action was run."
    return None

def private_write(path, value):
    temporary = path.with_name("." + path.name + "." + secrets.token_hex(6))
    try:
        with open(os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), "w") as handle:
            handle.write(value)
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(temporary, path)
        descriptor = os.open(path.parent, os.O_RDONLY)
        try:
            os.fsync(descriptor)
        finally:
            os.close(descriptor)
    finally:
        temporary.unlink(missing_ok=True)


def password_at(folder):
    path = folder / ".password"
    try:
        descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    except FileExistsError:
        descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
        with os.fdopen(descriptor) as handle:
            if not stat.S_ISREG(os.fstat(handle.fileno()).st_mode):
                raise RuntimeError("Password path must be a regular file")
            os.fchmod(handle.fileno(), 0o600)
            password = handle.read().strip()
        if len(password) < 24:
            raise RuntimeError("Password file must contain at least 24 characters")
        return password
    with os.fdopen(descriptor, "w") as handle:
        password = secrets.token_urlsafe(24)
        handle.write(password + "\n")
        handle.flush()
        os.fsync(handle.fileno())
    return password

    def launch(self, identifier):
        # Caller holds self.lock; the same saved assistant message is reused.
        if not self.closing.is_set():
            cancel = threading.Event()
            worker = threading.Thread(target=self.complete, args=(identifier, cancel), daemon=True)
            self.workers[identifier] = {"thread": worker, "cancel": cancel, "started_at": time.time()}
            worker.start()

    def complete(self, identifier, cancel):
        with self.lock:
            conversation = copy.deepcopy(self.get(identifier))
            value = self.get(identifier)
            if value["messages"] and value["messages"][-1]["status"] == "queued":
                value["messages"][-1].update(status="running", provider=None, attempts=[])
                self.save(value)
        def progress(provider, attempts):
            with self.lock:
                value = self.get(identifier)
                value["messages"][-1].update(status="running", provider=provider, attempts=attempts)
                self.save(value)
        try:
            answer = self.providers.reply(conversation, cancel, progress)
            if cancel.is_set():
                raise Cancelled("Reply stopped.")
            with self.lock:
                value = self.get(identifier)
                value["codex_thread_id"] = answer.pop("codex_thread_id", None)
                value.pop("retry_prompt", None)
                answer["content"] = self.redact(answer["content"])
                value["messages"][-1].update(answer, status="done")
                value["messages"][-1].pop("retry_at", None)
                self.save(value)
                user = next((m["content"] for m in reversed(value["messages"]) if m["role"] == "user"), "")
                self.record_turn(ask=user, provider=answer.get("provider"),
                            jobs=[], files=[], commits=[], confirmations=[], refusals=[],
                            open_items=[user[:200]] if user else [])
        except AllDown as exc:
            with self.lock:
                value = self.get(identifier)
                value["codex_thread_id"] = None
                value["retry_prompt"] = exc.prompt
                value["messages"][-1].update(status="cancelled" if cancel.is_set() else "waiting",
                    provider=None, content="Reply stopped." if cancel.is_set() else str(exc),
                    attempts=exc.attempts, retry_at=exc.retry_at)
                self.save(value)
        except Exception as exc:
            with self.lock:
                value = self.get(identifier)
                value["codex_thread_id"] = None
                value.pop("retry_prompt", None)
                message = str(exc) if isinstance(exc, ProviderError) else "The reply failed unexpectedly. Your message is saved; please try again."
                value["messages"][-1].update(status="cancelled" if isinstance(exc, Cancelled) else "error", content=self.redact(message))
                self.save(value)
        finally:
            with self.lock:
                self.workers.pop(identifier, None)

    def stop(self, identifier):
        with self.lock:
            value = self.get(identifier)
            info = self.workers.get(identifier)
            if info:
                info["cancel"].set()
            elif value["messages"] and value["messages"][-1]["status"] == "waiting":
                value["messages"][-1].update(status="cancelled", content="Queued request stopped.")
                value.pop("retry_prompt", None)
                self.save(value)
                if hasattr(self.providers, "circuit"):
                    self.providers.circuit.log("queue_cancelled", conversation_id=identifier)

    def retry_queued(self):
        with self.lock:
            self.sweep_stuck()
            queued = sorted((c for c in self.conversations.values() if c["messages"] and c["messages"][-1]["status"] == "waiting"), key=lambda c: c["messages"][-1]["created_at"])
            for value in queued:
                if value["id"] in self.workers:
                    continue
                message = value["messages"][-1]
                if message.get("retry_at", 0) <= time.time():
                    message.update(status="queued", content="", provider=None)
                    self.save(value)
                    if hasattr(self.providers, "circuit"):
                        self.providers.circuit.log("queue_retry", conversation_id=value["id"])
                    self.launch(value["id"])
                    return

    def start_background(self):
        self.providers.start_probes()
        def loop():
            while not self.closing.wait(1):
                try:
                    if not self.closing.is_set():
                        self.retry_queued()
                        self.sweep_stuck()
                except Exception:
                    self.providers.circuit.log("queue_error", reason="queued retry could not start")
        self.queue_thread = threading.Thread(target=loop, daemon=True)
        self.queue_thread.start()

    def close(self):
        self.closing.set()
        with self.lock:
            for info in self.workers.values():
                info["cancel"].set()
        if self.queue_thread:
            self.queue_thread.join(timeout=3)
        for info in list(self.workers.values()):
            info["thread"].join(timeout=10)
        if hasattr(self.providers, "close"):
            self.providers.close()

class Handler(BaseHTTPRequestHandler):
    server_version = "BobWebchat/1"

    def setup(self):
        super().setup()
        self.connection.settimeout(10)

    def log_message(self, *_args):
        pass  # No URLs, cookies, prompts, passwords or provider output in HTTP logs.

    def send(self, status, data, content_type="application/json; charset=utf-8", headers=None):
        if isinstance(data, (dict, list)):
            data = json.dumps(data, ensure_ascii=False)
        if isinstance(data, str):
            data = data.encode()
        self.send_response(status)
        for key, value in {
            "Content-Type": content_type, "Content-Length": str(len(data)), "Cache-Control": "no-store",
            "X-Content-Type-Options": "nosniff", "X-Frame-Options": "DENY", "Referrer-Policy": "no-referrer",
            "Content-Security-Policy": "default-src 'none'; script-src 'self'; style-src 'self'; img-src 'self'; media-src 'self'; connect-src 'self'; form-action 'self'; frame-ancestors 'none'; base-uri 'none'",
            "X-Bob-Webchat": "1", **(headers or {}),
        }.items():
            self.send_header(key, value)
        self.end_headers()
        self.wfile.write(data)

    def body(self, form=False, limit=None):
        if self.headers.get("Transfer-Encoding"):
            raise RequestError(400, "Transfer encoding is not supported.")
        cap = limit or MAX_BODY
        try:
            size = int(self.headers.get("Content-Length", "0"))
        except ValueError:
            raise RequestError(400, "Invalid content length.") from None
        if not 0 < size <= cap:
            raise RequestError(413, "Request is empty or too large.")
        mime = self.headers.get("Content-Type", "").split(";", 1)[0].strip().lower()
        if mime != ("application/x-www-form-urlencoded" if form else "application/json"):
            raise RequestError(415, "Unsupported content type.")
        try:
            raw = self.rfile.read(size).decode("utf-8")
            data = {k: v[0] for k, v in parse_qs(raw).items()} if form else json.loads(raw)
            if not isinstance(data, dict):
                raise ValueError()
            return data
        except (UnicodeError, ValueError):
            raise RequestError(400, "Invalid request body.") from None

    def cookie(self, token, clear=False):
        secure = self.origin.startswith("https://")
        return f"{COOKIE}={token}; Path=/; HttpOnly; SameSite=Strict; Max-Age={0 if clear else SESSION_SECONDS}" + ("; Secure" if secure else "")

    def dispatch(self):
        try:
            hosts = self.headers.get_all("Host", [])
            if len(hosts) != 1 or hosts[0].lower() not in self.server.origins:
                raise RequestError(403, "Unrecognized host.")
            self.origin = self.server.origins[hosts[0].lower()]
            self.path_only = urlsplit(self.path).path
            session = self.server.chat.authenticate(self.headers.get("Cookie"))
            if self.path_only == "/api" or self.path_only.startswith("/api/"):
                if not session:
                    raise RequestError(401, "Please log in.")
            if self.command == "POST":
                origins = self.headers.get_all("Origin", [])
                if (origins and origins != [self.origin]) or self.headers.get("Sec-Fetch-Site") in ("cross-site", "same-site"):
                    raise RequestError(403, "Cross-site requests are not allowed.")
                if self.path_only.startswith("/api/"):
                    csrf = self.headers.get("X-CSRF-Token", "")
                    if not hmac.compare_digest(csrf.encode(), session["csrf"].encode()):
                        raise RequestError(403, "Invalid request token. Reload the page.")
                self.post(session)
            elif self.command == "GET":
                self.get(session)
            else:
                raise RequestError(405, "Method not allowed.")
        except RequestError as exc:
            self.send(exc.status, {"error": exc.message})
        except (BrokenPipeError, ConnectionResetError, TimeoutError):
            pass
        except Exception:
            self.send(500, {"error": "The request failed. Please try again."})

    do_POST = dispatch
    do_OPTIONS = dispatch
    do_PUT = dispatch
    do_DELETE = dispatch
    do_PATCH = dispatch

def main():
    os.umask(0o077)
    with (HERE / ".server.lock").open("a") as lock:
        try:
            fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            raise SystemExit("Bob web chat is already running.")
        chat = Chat()
        origin_path = HERE / ".public-origin"
        origin = origin_path.read_text().strip() if origin_path.exists() else ""
        server = Server((HOST, PORT), chat, origin)
        chat.start_background()
        def shutdown(_signum, _frame):
            with chat.lock:
                for info in chat.workers.values():
                    info["cancel"].set()
            threading.Thread(target=server.shutdown, daemon=True).start()
        for signum in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP):
            signal.signal(signum, shutdown)
        print(f"Bob web chat listening on http://{HOST}:{PORT}; password: {HERE / '.password'}", flush=True)
        try:
            server.serve_forever(poll_interval=0.25)
        finally:
            server.server_close()
            chat.close()


if __name__ == "__main__":
    main()

  $("logout").addEventListener("click", async () => {
    try { await api("/api/logout", {}); location.assign("/login"); }
    catch (error) { $("notice").textContent = error.message; }
  });
  for (const button of document.querySelectorAll("[data-report]")) button.addEventListener("click", async () => {
    $("report-title").textContent = button.textContent;
    $("report-body").textContent = "Loading…"; $("report-date").textContent = ""; $("report").showModal();
    try {
      const data = await api(`/api/${button.dataset.report}`);
      $("report-date").textContent = data.modified_at ? `File updated ${new Date(data.modified_at * 1000).toLocaleString()} · saved report, not a new check` : `Read ${new Date(data.read_at * 1000).toLocaleString()}`;
      $("report-body").textContent = button.dataset.report === "jobs" ?
        `${data.maintenance ? "Scheduling paused: " + data.maintenance + "\n\n" : ""}SCREENS\n${data.screens}\n\nQUEUE${data.queue.modified_at ? " · updated " + new Date(data.queue.modified_at * 1000).toLocaleString() : ""}\n${data.queue.text}` :
        data.text + (data.truncated ? "\n\n[Display truncated]" : "");
    } catch (error) { $("report-body").textContent = error.message; }
  });
  $("close-report").addEventListener("click", () => $("report").close());
  let songProjectId = stored("bob.songProject") || "";
  async function loadSong() {
    try {
      const list = await api("/api/song-projects");
      const row = list.projects.find((p) => p.id === songProjectId) || list.projects[0];
      if (!row) { $("song-summary").textContent = "new"; return; }
      songProjectId = row.id; remember("bob.songProject", songProjectId);
      const project = await api(`/api/song-projects/${songProjectId}`);
      $("song-title").value = project.title || "";
      $("song-lyrics").value = project.lyrics || "";
      $("song-chords").value = project.chords || "";
      $("song-notes").value = project.notes || "";
      $("song-summary").textContent = project.title || "saved";
    } catch (_) { $("song-summary").textContent = "offline"; }
  }
  $("song-save").addEventListener("click", async () => {
    $("song-notice").textContent = "Saving…";
    try {
      const project = await api("/api/song-projects", {
        id: songProjectId || undefined,
        title: $("song-title").value,
        lyrics: $("song-lyrics").value,
        chords: $("song-chords").value,
        notes: $("song-notes").value,
        version_note: "saved from phone",
      });
      songProjectId = project.id; remember("bob.songProject", songProjectId);
      $("song-notice").textContent = "Saved " + new Date().toLocaleTimeString();
      $("song-summary").textContent = project.title;
    } catch (error) { $("song-notice").textContent = error.message; }
  });
  $("song-demo").addEventListener("click", async () => {
    const lyrics = $("song-lyrics").value.trim();
    if (!lyrics) { $("song-notice").textContent = "Add lyrics first."; return; }
    let receipt = "";
    try {
      const quote = await api("/api/music/demo", { lyrics, style: $("song-notes").value || "acoustic songwriter" });
      if (quote.needs_confirm) {
        receipt = quote.receipt || "";
        if (!quote.can_afford_demo) { $("song-notice").textContent = quote.message || "Cap too low for a demo."; return; }
        if (!window.confirm(quote.message || "Generate MiniMax demo?")) return;
      }
    } catch (error) { $("song-notice").textContent = error.message; return; }
    $("song-notice").textContent = "Generating demo…";
    try {
      const demo = await api("/api/music/demo", {
        lyrics, style: $("song-notes").value || "acoustic songwriter", confirm: true, receipt,
      });
      if (demo.audio_id) {
        const player = $("song-player");
        player.hidden = false;
        player.src = `/api/audio/${demo.audio_id}`;
        $("song-notice").textContent = `Demo ready · ${demo.engine || "local-sketch"} · ${demo.bytes} bytes`;
      } else $("song-notice").textContent = demo.message || "Demo queued.";
    } catch (error) { $("song-notice").textContent = error.message; }
  });
  $("song-upload").addEventListener("change", async (event) => {
    const file = event.target.files?.[0];
    if (!file) return;
    $("song-notice").textContent = "Uploading…";
    try {
      const buffer = await file.arrayBuffer();
      const bytes = new Uint8Array(buffer);
      let binary = "";
      for (let i = 0; i < bytes.length; i += 0x8000) binary += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
      const stored = await api("/api/audio/upload", { filename: file.name, data_base64: btoa(binary) });
      const player = $("song-player");
      player.hidden = false;
      player.src = `/api/audio/${stored.audio_id}`;
      $("song-notice").textContent = `Uploaded ${file.name}`;
    } catch (error) { $("song-notice").textContent = error.message; }
    event.target.value = "";
  });
  async function refreshPanel() {
    try {
      const data = await api("/api/panel");
      const state = data.handback?.bob_state?.text || "";
      $("handback-chip").textContent = state ? state.split("\n").find((line) => line.trim() && !line.startsWith("#"))?.slice(0, 140) || state.slice(0, 140) : "Handback: no BOB-STATE yet";
      const dash = data.dashboard_modified_at ? new Date(data.dashboard_modified_at * 1000).toLocaleString() : "unknown";
      const ops = data.live_ops || {};
      const spend = ops.minimax_spend || {};
      const roles = ops.roles || {};
      const jobs = (ops.running_jobs || []).slice(0, 4);
      $("status-summary").textContent = `round ${ops.webchat_round || "?"} · MiniMax $${(spend.committed_usd ?? 0).toFixed(2)}/${spend.cap_usd || 59}`;
      const scoreLines = (ops.scores || []).slice(0, 6).map((row) => `${row.area}: ${row.score}`).join(" · ");
      const jobLines = jobs.map((job) => {
        const commit = job.last_commit?.sha ? ` · ${job.last_commit.sha}` : "";
        const mins = Math.round((job.elapsed_seconds || 0) / 60);
        return `• ${job.engine || "job"} ${job.stall_state} · ${mins}m · ${job.plain_now || "working"}${commit}`;
      });
      const lines = [
        `Dashboard fresh: ${dash}`,
        `Builder: ${roles.building || "idle"} · Review: ${roles.reviewing || "idle"}`,
        scoreLines ? `Scores: ${scoreLines}` : "",
        jobLines.length ? "Live jobs:\n" + jobLines.join("\n") : "Live jobs: none on screen",
        "",
        (data.handback?.summary?.text || data.handback?.bob_state?.text || "").slice(0, 900),
      ].filter(Boolean);
      $("status-body").textContent = lines.join("\n");
    } catch (_) {
      $("status-summary").textContent = "unavailable";
    }
  }
  async function refreshLanes() {
    try {
      const data = await api("/api/lanes" + (current ? "?conversation_id=" + encodeURIComponent(current) : ""));
      const entries = Object.entries(data.providers);
      $("lanes-summary").textContent = entries.filter(([, lane]) => lane.state === "cooldown").length + " cooling down";
      $("lanes").replaceChildren(...entries.map(([name, lane]) => {
        const row = document.createElement("div"); row.className = "lane";
        const title = document.createElement("strong");
        title.textContent = `${providers[name]} · ${lane.state.replace("_", " ")}${lane.automatic ? "" : " · explicit choice only"}`;
        const detail = document.createElement("span");
        const spend = lane.spend;
        detail.textContent = lane.detail + (lane.retry_at > Date.now() / 1000 ? ` · retry ${new Date(lane.retry_at * 1000).toLocaleTimeString()}` : "") +
          (spend.committed_usd !== undefined && ["gemini", "minimax"].includes(name) ? ` · $${spend.committed_usd.toFixed(4)} committed${spend.cap_usd ? " / $" + spend.cap_usd : ""} (includes reservations)` : ` · ${spend.billing}`) +
          (lane.probed_at ? ` · probe ${lane.probe}, ${new Date(lane.probed_at * 1000).toLocaleTimeString()} (login/configuration/network only)` : "");
        row.append(title, detail); return row;
      }));
    } catch (_) { $("lanes-summary").textContent = "Health unavailable"; }
  }
  async function init() {
    try {
      csrf = (await api("/api/session")).csrf; current = stored("bob.conversation");
      await refreshList(); await poll(); $("messages").scrollTop = $("messages").scrollHeight;
      await refreshPanel(); await refreshLanes(); await loadSong();
      setInterval(refreshPanel, 8000);
      setInterval(refreshLanes, 5000);
      setInterval(poll, 1200);
    } catch (error) { $("notice").textContent = error.message; }
  }
  composerState(); init();
